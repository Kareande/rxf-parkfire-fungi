  --input-path manifest \
  --output-path seqs_demux.qza \

## NOTES:

# but poor quality <30 for reverse reads. Dumping reverse read and only processing as single justified,
# but chose to run pipeline with paired reads to compare.


#----------------------------------------------------------------------------------------------------------------------#
#-------------------------------------------------- TRIMMING PRIMERS --------------------------------------------------#
#----------------------------------------------------------------------------------------------------------------------#
# kubectl create -f Qiime/YAMLs/job-cutadapt.yaml

# Trim the f-primer and the reverse complement of the r-primer from the f-reads
qiime cutadapt trim-paired \
  --i-demultiplexed-sequences seqs_demux.qza \
  --p-front-f CTTGGTCATTTAGAGGAAGTAA \
  --p-front-r GCTGCGTTCTTCATCGATGC \
  --p-error-rate 0.13 \
  --p-cores 8 \
  --p-discard-untrimmed \
  --o-trimmed-sequences cutadapt/seqs_demux_trimmed.qza \
  --o-stats cutadapt/stats_cutadapt_sgl.qza

# Summarize trimmed data
qiime demux summarize \
 --i-data cutadapt/seqs_demux_trimmed.qza \
 --o-visualization cutadapt/seqs_demux_trimmed.qzv

## Download above summaries for visualization (https://view.qiime2.org/)
kubectl cp grad-erth-wmis-kranderson4/qiime2-pod:/biovol/cutadapt/seqs_demux_trimmed.qzv Qiime/visualizations/cutadapt/seqs_demux_trimmed.qzv

## NOTES: 
# You can see evidence that the first 22bp were trimmed by looking at the trimmed seqs interactive quality plot.
# N samples: 225
# Median reads per sample: 9559

#---------------------------------------------- QUALITY FILTER & DENOISE ----------------------------------------------#
#----------------------------------------------------------------------------------------------------------------------#
# kubectl create -f Qiime/YAMLs/job-cutadapt-dada2-single.yaml

## Use dada2 to denoise, dereplicate, and filter chimeras
qiime dada2 denoise-single \
  --i-demultiplexed-seqs cutadapt/seqs_demux_trimmed.qza \
  --p-trim-left 0 \
  --p-trunc-len 0 \
  --o-representative-sequences cutadapt/seqs_asvs_sgl.qza \
  --o-table cutadapt/table_asvs_sgl.qza \
  --o-denoising-stats cutadapt/stats_dada2_sgl.qza \
  --o-base-transition-stats cutadapt/stats_dada2_btrans_sgl.qza \
  --p-n-threads 0

## Get sample and feature seq freq and summary stats
  qiime metadata tabulate \
  --m-input-file cutadapt/stats_dada2_sgl.qza  \
  --o-visualization cutadapt/stats_dada2_sgl.qzv
qiime feature-table summarize \
  --i-table cutadapt/table_asvs_sgl.qza \
  --m-metadata-file metadata.tsv \
  --o-feature-frequencies cutadapt/feat_freq_asvs_sgl.qza \
  --o-sample-frequencies cutadapt/smpl_freq_asvs_sgl.qza \
  --o-summary cutadapt/table_asvs_sgl.qzv
qiime feature-table tabulate-seqs \
  --i-data cutadapt/seqs_asvs_sgl.qza \
  --o-visualization cutadapt/seqs_asvs_sgl.qzv
