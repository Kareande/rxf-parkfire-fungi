#------------------------------------------------- FILTERING BY READS -------------------------------------------------#
#----------------------------------------------------------------------------------------------------------------------#
# job-filter.yaml

## Export to a TSV, this contains the counts
qiime tools export \
--input-path seqs_demux.qzv \
--output-path demux_summary_export

## Filter out samples w/ <500 reads
qiime demux filter-samples \
  --i-demux seqs_demux.qza \
  --m-metadata-file demux_summary_export/per-sample-fastq-counts.tsv \
  --p-where 'CAST([forward sequence count] AS INT) > 500' \
  --p-remove-empty \
  --o-filtered-demux seqs_filt.qza \
  --verbose

## Verify
qiime demux summarize \
  --i-data seqs_filt.qza \
  --o-visualization seqs_filt.qzv

## Download above summaries for visualization (https://view.qiime2.org/)
kubectl cp grad-erth-wmis-kranderson4/qiime2-pod:/biovol/seqs_filt.qzv seqs_filt.qzv

## NOTE:
# Filtering of low read samples was necessary in order to use ITSxpress. Alternatively,
# I could have chosen to simply trim the primers using cutadapt. I compared the two, and
# ITSxpress results were much better (higher quality seqs, more features).

#------------------------------------------------- EXTRACT ITS REGION -------------------------------------------------#
#----------------------------------------------------------------------------------------------------------------------#
# kubectl create -f job-itsxpress.yaml

## Install itsxpress
conda install -c bioconda -c conda-forge ITSxpress
qiime dev refresh-cache

## Extract ITS sequencing from raw data, no clustering
qiime itsxpress trim-pair-output-unmerged --verbose \
--i-per-sample-sequences seqs_filt.qza \
--p-region ITS1 \
--p-threads 0 \
--p-cluster-id 1 \
--o-trimmed seqs_its_extracted.qza

## Summarize seqs for visualization
qiime demux summarize \
  --i-data seqs_its_extracted.qza \
  --o-visualization seqs_its_extracted.qzv

## Download above summaries for visualization (https://view.qiime2.org/)
kubectl cp grad-erth-wmis-kranderson4/qiime2-pod:/biovol/seqs_its_extracted.qzv seqs_its_extracted.qzv

## NOTES: 
# You can see evidence that the ITS1 region was extracted by looking at the trimmed seqs
# interactive quality plot.


#----------------------------------------------------------------------------------------------------------------------#
#---------------------------------------------- QUALITY FILTER & DENOISE ----------------------------------------------#
#----------------------------------------------------------------------------------------------------------------------#
# kubectl create -f job-dada2-single.yaml

## Use dada2 to denoise, dereplicate, and filter chimeras
qiime dada2 denoise-single \
  --i-demultiplexed-seqs seqs_its_extracted.qza \
  --p-trim-left 0 \
  --p-trunc-len 0 \
  --o-representative-sequences asvs_dada2_single.qza \
  --o-table table_dada2_single.qza \
  --o-denoising-stats stats_dada2_single.qza \
  --o-base-transition-stats bt_stats_dada2_single.qza \
  --p-n-threads 0

  qiime metadata tabulate \
  --m-input-file stats_dada2_single.qza  \
  --o-visualization stats_dada2_single.qzv
qiime feature-table summarize \
  --i-table table_dada2_single.qza \
  --m-metadata-file metadata.tsv \
  --o-feature-frequencies feat_freq_dada2_single.qza \
  --o-sample-frequencies smpl_freq_dada2_single.qza \
  --o-summary table_dada2_single.qzv
qiime feature-table tabulate-seqs \
  --i-data asvs_dada2_single.qza \
  --o-visualization asvs_dada2_single.qzv

## Download above summaries for visualization (https://view.qiime2.org/)

#----------------------------------------------------------------------------------------------------------------------#
#--------------------------------------------------- DECONTAMINATE ----------------------------------------------------#
#----------------------------------------------------------------------------------------------------------------------#
# https://jordenrabasco.github.io/Q2_Decontam_Tutorial.html
# job-decontam.yaml

## Upload mock community for Filtering
kubectl cp data\mock_sequences.fasta qiime2-pod:\biovol\mock_sequences.fasta

## Remove mock community from pos ctls
qiime tools import \
  --type 'FeatureData[Sequence]' \
  --input-path mock_sequences.fasta \
  --output-path mock_seqs.qza
qiime feature-table filter-features \
  --i-table table_dada2_single.qza \
  --m-metadata-file mock_seqs.qza \
  --p-exclude_ids \
  --o-filtered-table table_dada2_single_filt.qza

## Identify contaminant seqs (prevalence method)
qiime quality-control decontam-identify \
  --i-table table_dada2_single_filt.qza \
  --m-metadata-file metadata.tsv \
  --p-method prevalence \
  --p-prev-control-column Sample_or_Control \
  --p-prev-control-indicator control \
  --o-decontam-scores decontam_pvlnc_scores_unclst_sgl.qza

## Visualize the decontam scores
qiime quality-control decontam-score-viz \
  --i-decontam-scores decontam_pvlnc_scores_unclst_sgl.qza \
  --i-table table_dada2_single.qza \
  --i-rep-seqs asvs_dada2_single.qza \
  --p-threshold 0.1 \
  --p-no-weighted \
  --p-bin-size 0.05 \
  --o-visualization decontam_pvlnc_scores_unclst_sgl.qzv

## Filter out contaminants from table
qiime feature-table filter-features \
  --i-table table_dada2_single.qza \
  --m-metadata-file decontam_pvlnc_scores_unclst_sgl.qza \
  --p-where '[p]>0.1 OR [p] IS NULL' \
  --o-filtered-table table_pdecont_unclst_sgl.qza

## Filter out contaminants from seqs
qiime feature-table filter-seqs \
  --i-data asvs_dada2_single.qza \
  --i-table table_pdecont_unclst_sgl.qza \
  --o-filtered-data asvs_pdecont_sgl.qza

## Get sample and feature seq freq and summary stats
qiime feature-table summarize \
  --i-table table_pdecont_unclst_sgl.qza \
  --m-metadata-file metadata.tsv \
  --o-feature-frequencies feat_freq_pdecont_unclst_sgl.qza \
  --o-sample-frequencies smpl_freq_pdecont_unclst_sgl.qza \
  --o-summary table_pdecont_unclst_sgl.qzv

## Visualize feature IDs w/ BLAST links
qiime feature-table tabulate-seqs \
  --i-data asvs_pdecont_sgl.qza \
  --m-metadata-file feat_freq_pdecont_unclst_sgl.qza \
  --o-visualization asvs_pdecont_sgl.qzv

## Download the visualization
kubectl cp grad-erth-wmis-kranderson4/qiime2-pod:/biovol/decontam_pvlnc_scores_unclst_sgl.qzv decontam_pvlnc_scores_unclst_sgl.qzv
kubectl cp grad-erth-wmis-kranderson4/qiime2-pod:/biovol/table_pdecont_unclst_sgl.qzv table_pdecont_unclst_sgl.qzv
kubectl cp grad-erth-wmis-kranderson4/qiime2-pod:/biovol/asvs_pdecont_sgl.qzv asvs_pdecont_sgl.qzv

## NOTES: 
# The combined decontamination filtered out 30 ASVs and 8.5mil seqs (out of 27mil). The prevalence
# method filtered out 134 ASVs and 8.5mil seqs. I also ran both decontam methods on the unclustered
# sequences, and there was little difference between methods, but there are twice as many ASVs as
# the clustered though the total seqs (18-19mil) ended up being similar to the 97% clustered data.
# The mean frequences per sample are all similar (~117k and ~120k), while the frequency per feature
# is of course higher in the clustered data where ASVs were collapsed together (~7k and ~4.5k).

#----------------------------------------- ASSIGN TAXONOMY - TRAIN CLASSIFIER -----------------------------------------#
#----------------------------------------------------------------------------------------------------------------------#
#job-classifier-UNITE-training.yaml

## Upload latest UNITE data and associated taxonomy file (w/ mock community data added)
kubectl cp sh_qiime_release_s_19.02.2025\sh_refs_qiime_ver10_dynamic_s_19.02.2025.fasta qiime2-pod:\biovol\sh_refs_qiime_ver10_dynamic_s_19.02.2025.fasta
kubectl cp sh_qiime_release_s_19.02.2025\sh_taxonomy_qiime_ver10_dynamic_s_19.02.2025.txt qiime2-pod:\biovol\sh_taxonomy_qiime_ver10_dynamic_s_19.02.2025.txt

## Import latest UNITE data
qiime tools import \
  --type 'FeatureData[Sequence]' \
  --input-path sh_refs_qiime_ver10_dynamic_s_19.02.2025.fasta \
  --output-path unite.qza

## Import associated UNITE taxonomy file
qiime tools import \
--type 'FeatureData[Taxonomy]' \
--input-format HeaderlessTSVTaxonomyFormat \
--input-path sh_taxonomy_qiime_ver10_dynamic_s_19.02.2025.txt \
--output-path unite_taxonomy.qza

## Train classifier
qiime feature-classifier fit-classifier-naive-bayes \
  --i-reference-reads unite.qza \
  --i-reference-taxonomy unite_taxonomy.qza \
  --o-classifier classifier.qza

## Assign taxonomy to ASVs
qiime feature-classifier classify-sklearn \
  --i-classifier classifier.qza \
  --i-reads asvs_pdecon_unclst_noctl.qza \
  --o-classification asvs_taxaUNITE_unclst_trained.qza

## Visualize feature IDs and taxonomy w/ seqs linked to BLAST
qiime feature-table tabulate-seqs \
  --i-data asvs_pdecon_unclst_noctl.qza \
  --i-taxonomy asvs_taxaUNITE_unclst_trained.qza \
  --o-visualization asvs_taxaUNITE_unclst_trained.qzv

## Create barplot
qiime taxa barplot \
  --i-table table_pdecon_unclst_noctl.qza  \
  --i-taxonomy asvs_taxaUNITE_unclst_trained.qza \
  --m-metadata-file metadata_noctl.tsv \
  --o-visualization barplots_taxaUNITE_unclst_trained.qzv

#----------------------------------------------------------------------------------------------------------------------#
#---------------------------------------------- EXPORT DATA FROM QIIME ------------------------------------------------#
#----------------------------------------------------------------------------------------------------------------------#

## Export freq table as .biom (feature-table.biom: #OTU ID, taxonomy, confidence)
qiime tools export \
  --input-path table_pdecon_unclst_noctl.qza \
  --output-path export-R/freq-table

## Export fungal traits to .tsv (metadata.tsv: feature-id, taxon, confidence, traits)
qiime tools export \
  --input-path fungal_traits.qza \
  --output-path export-R/traits

## Download fungal traits data to use in R
kubectl cp grad-erth-wmis-kranderson4/qiime2-pod:/biovol/export-R/traits/metadata.tsv ./fungal_traits.tsv

## Replace header of trait table so that the tables match
# "feature-id  taxon confidence" -> "#OTU ID  taxonomy  confidence"
nano export-R/traits/metadata.tsv

## Add taxon metadata to the ASV freq table .biom (using trait metadata)
biom add-metadata \
 -i export-R/freq-table/feature-table.biom \
 --observation-metadata-fp export-R/traits/metadata.tsv \
 -o export-R/freq-table/feat_freq_taxonomy.biom \
 --sc-separated taxonomy

## Convert freq-taxon .biom to .tsv (feat_freq_taxonomy.tsv: #OTU ID, 10B1, 10C1, ..., taxonomy)
biom convert \
  -i export-R/freq-table/feat_freq_taxonomy.biom \
  -o export-R/freq-table/feat_freq_taxonomy.tsv \
  --to-tsv \
  --process-obs-metadata taxonomy \
  --header-key taxonomy

## Download freq-taxon data to use in R
kubectl cp grad-erth-wmis-kranderson4/qiime2-pod:/biovol/export-R/freq-table/feat_freq_taxonomy.tsv ./feat_freq_taxonomy.tsv

## Add sequences and fungal traits to freq table
nano export-R/freq-table/feat_freq_taxonomy.tsv #"FEATURE ID"
qiime feature-table tabulate-seqs \
  --i-data asvs_pdecon_unclst_noctl.qza \
  --m-metadata-file fungal_traits.qza \
  --m-metadata-file export-R/freq-table/feat_freq_taxonomy.tsv \
  --o-visualization asvs_pdecon_traits.qzv

## Download freq-seq-trait-taxon visualization (copy and paste from https://view.qiime2.org/)
