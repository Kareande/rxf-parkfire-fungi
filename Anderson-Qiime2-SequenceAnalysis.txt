## Import data into quiime2
qiime tools import \
  --type 'SampleData[PairedEndSequencesWithQuality]' \
  --input-path manifest \
  --output-path seqs_demux.qza \
  --input-format PairedEndFastqManifestPhred33V2

## summarize the data
qiime demux summarize \
  --i-data seqs_demux.qza \
  --o-visualization seqs_demux.qzv

## Download above summary for visualization (https://view.qiime2.org/)

## NOTES: 
# Interactive quality plot for demux seqs shows steady quality in forward reads (no truncation necessary),
# but poor quality <30 for reverse reads. Dumping reverse read and only processing as single justified,
# but chose to run pipeline with paired reads to compare.


#----------------------------------------------------------------------------------------------------------------------#
#------------------------------------------------- FILTERING BY READS -------------------------------------------------#
#----------------------------------------------------------------------------------------------------------------------#
# kubectl create -f Qiime/YAMLs/job-filter.yaml

## Export to a TSV, this contains the counts
qiime tools export \
--input-path seqs_demux.qzv \
--output-path demux_summary_export

## Filter out samples w/ <500 reads
qiime demux filter-samples \
  --i-demux seqs_demux.qza \
  --m-metadata-file demux_summary_export/per-sample-fastq-counts.tsv \
  --p-where 'CAST([forward sequence count] AS INT) > 500' \
  --p-remove-empty \
  --o-filtered-demux seqs_demux_filt.qza \
  --verbose

## Verify
qiime demux summarize \
  --i-data seqs_demux_filt.qza \
  --o-visualization seqs_demux_filt.qzv

## Download above summaries for visualization (https://view.qiime2.org/)
kubectl cp grad-erth-wmis-kranderson4/qiime2-pod:/biovol/seqs_demux_filt.qzv Qiime/visualizations/seqs_demux_filt.qzv

## NOTE:
# Filtering of low read samples was necessary in order to use ITSxpress. Alternatively,
# I could have chosen to simply trim the primers using cutadapt. I compared the two, and
# ITSxpress results were much better (higher quality seqs, more features).

#----------------------------------------------------------------------------------------------------------------------#
#------------------------------------------------- EXTRACT ITS REGION -------------------------------------------------#
#----------------------------------------------------------------------------------------------------------------------#
# kubectl create -f Qiime/YAMLs/job-itsxpress.yaml

## Install itsxpress
conda install -c bioconda -c conda-forge ITSxpress
qiime dev refresh-cache

## Extract ITS sequencing from raw data
qiime itsxpress trim-pair-output-unmerged --verbose \
--i-per-sample-sequences seqs_demux_filt.qza \
--p-region ITS1 \
--p-threads 0 \
--p-cluster-id 1 \
--o-trimmed seqs_demux_trimmed.qza

## Summarize seqs for visualization
qiime demux summarize \
  --i-data seqs_demux_trimmed.qza \
  --o-visualization seqs_demux_trimmed.qzv

## Download above summaries for visualization (https://view.qiime2.org/)
kubectl cp grad-erth-wmis-kranderson4/qiime2-pod:/biovol/seqs_demux_trimmed.qzv Qiime/visualizations/seqs_demux_trimmed.qzv

## NOTES: 
# You can see evidence that the ITS1 region was extracted by looking at the trimmed seqs interactive quality plot. There
# N samples: 157
# Median f-reads per sample: 14697


#----------------------------------------------------------------------------------------------------------------------#
#---------------------------------------------- QUALITY FILTER & DENOISE ----------------------------------------------#
#----------------------------------------------------------------------------------------------------------------------#
# kubectl create -f Qiime/YAMLs/job-dada2-single.yaml

## Use dada2 to denoise, dereplicate, and filter chimeras
qiime dada2 denoise-single \
  --i-demultiplexed-seqs seqs_demux_trimmed.qza \
  --p-trim-left 0 \
  --p-trunc-len 0 \
  --o-representative-sequences seqs_asvs_sgl.qza \
  --o-table table_asvs_sgl.qza \
  --o-denoising-stats stats_dada2_sgl.qza \
  --o-base-transition-stats stats_dada2_btrans_sgl.qza \
  --p-n-threads 0

## Get sample and feature seq freq and summary stats
  qiime metadata tabulate \
  --m-input-file stats_dada2_sgl.qza  \
  --o-visualization stats_dada2_sgl.qzv
qiime feature-table summarize \
  --i-table table_asvs_sgl.qza \
  --m-metadata-file metadata.tsv \
  --o-feature-frequencies feat_freq_asvs_sgl.qza \
  --o-sample-frequencies smpl_freq_asvs_sgl.qza \
  --o-summary table_asvs_sgl.qzv
qiime feature-table tabulate-seqs \
  --i-data seqs_asvs_sgl.qza \
  --o-visualization seqs_asvs_sgl.qzv

  --i-table table_asvs_sgl.qza \
